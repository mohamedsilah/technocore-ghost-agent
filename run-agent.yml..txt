name: Run Technocore Ghost Agent

on:
  workflow_dispatch:
  schedule:
    - cron: '0 * * * *'

jobs:
  run-agent:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout Code
        uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: 20

      - name: Install Dependencies
        run: |
          npm cache clean --force
          npm install

      - name: Execute Ghost Agent
        run: node ghost-agent.js
        env:
          TECHNOCORE_URL: https://technocore.chat
          TECHNOCORE_SIGNING_KEY: ${{ secrets.TECHNOCORE_SIGNING_KEY }}
