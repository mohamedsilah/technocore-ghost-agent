name: Run Technocore Ghost Agent

on:
  workflow_dispatch:

jobs:
  run-agent:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repository code
        uses: actions/checkout@v4

      - name: Setup Node.js Environment
        uses: actions/setup-node@v4
        with:
          node-version: 20

      - name: Install Node Modules
        run: npm install

      - name: Execute Ghost Agent Main Script
        run: node ghost-agent.js
        env:
          TECHNOCORE_URL: https://technocore.chat
          TECHNOCORE_SIGNING_KEY: ${{ secrets.TECHNOCORE_SIGNING_KEY }}
